# stock_analysis 常用命令

## 环境准备
pip install -e .                        # 安装依赖
cp .env.example .env                    # 创建配置文件（需手动填入 API Key）
pytest tests/ -v                        # 运行测试 (30 tests)

## 分析命令

# 快速扫描（默认模式，最快）
python -m src.app.cli analyze --ticker 600519 --mode quick

# 深度分析（使用 deepseek-v4-pro，含知识库检索）
python -m src.app.cli analyze --ticker 600519 --mode deep

# 价值评估
python -m src.app.cli analyze --ticker 600519 --mode value

# 交易决策（含支撑/阻力/目标价/置信度）
python -m src.app.cli analyze --ticker 600519 --mode trade

# 无 LLM 模式（零 Token，仅输出 JSON 分析包）
python -m src.app.cli analyze --ticker 600519 --mode quick --no-llm

# 生成 K 线图（交互式 HTML）
python -m src.app.cli analyze --ticker 600519 --mode trade --chart

# 指定历史日期
python -m src.app.cli analyze --ticker 600519 --mode trade --date 2026-07-10

# 批量分析
python -m src.app.cli analyze --tickers 600519,000858,002837 --mode quick

# 调试模式（显示详细日志）
python -m src.app.cli analyze --ticker 600519 --mode trade --debug

## 多股票对比
python -m src.app.cli compare --tickers 600519,000858,002837
python -m src.app.cli compare --tickers 600519,000858,002837 --chart   # 含走势图

## 历史与统计
python -m src.app.cli history                        # 查看最近分析记录
python -m src.app.cli history --ticker 600519        # 按股票筛选
python -m src.app.cli stats                          # 统计信息
python -m src.app.cli cost                           # Token 成本统计
python -m src.app.cli modes                          # 列出所有模式

## 股票代码格式
# 支持完整格式：600519.SH, 000858.SZ, 002837.SZ
# 支持简写：600519（6开头自动补.SH，其他补.SZ）

## 输出文件
output/reports/          # Markdown 分析报告
output/json/             # 结构化分析包 JSON
output/charts/           # K 线图 HTML
output/history.json      # 分析历史记录
logs/                    # 运行日志

## 四种模式对比
# quick  - 快速扫描  | flash | ~3K tokens | 无知识库
# deep   - 深度分析  | pro   | ~6K tokens | 含知识库检索
# value  - 价值评估  | flash | ~3K tokens | 含知识库检索
# trade  - 交易决策  | flash | ~5K tokens | 含知识库检索 + 价格水平
